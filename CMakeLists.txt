cmake_minimum_required(VERSION 3.28)
project(qctsm VERSION 0.1.0 LANGUAGES C CXX)

file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" _qctsm_source)
file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" _qctsm_binary)
cmake_path(IS_PREFIX _qctsm_source "${_qctsm_binary}" NORMALIZE _qctsm_inside)
if(_qctsm_inside)
  message(FATAL_ERROR "QCTSM requires an external build directory")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_OBJECT_PATH_MAX 200 CACHE STRING "Portable generated object path budget")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  add_compile_options(/utf-8 /permissive- /Zc:__cplusplus)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum macOS")
  enable_language(OBJC OBJCXX)
  set(CMAKE_INSTALL_RPATH "@executable_path/../Frameworks")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(GNUInstallDirs)
if(APPLE)
  set(CMAKE_INSTALL_BINDIR "qctsm-gui.app/Contents/MacOS")
  set(CMAKE_INSTALL_LIBDIR "qctsm-gui.app/Contents/Frameworks")
  set(CMAKE_INSTALL_DATADIR "qctsm-gui.app/Contents/Resources")
endif()
include(CTest)
option(QCOPILOTS_TOOL_SEMANTIC_MATCHING_CORE_ONLY "Build only the QGIS-compatible retrieval SDK" OFF)
option(QCOPILOTS_TOOL_SEMANTIC_MATCHING_BUILD_TESTS "Build all product tests" ON)
option(QCOPILOTS_TOOL_SEMANTIC_MATCHING_DEPLOY_QT "Deploy Qt into the installed product" ON)
set(QCOPILOTS_TOOL_SEMANTIC_MATCHING_DEPENDENCY_ROOT "${CMAKE_BINARY_DIR}/../deps" CACHE PATH "External source and dependency cache")
set(QCOPILOTS_TOOL_SEMANTIC_MATCHING_TEST_ROOT "${CMAKE_BINARY_DIR}/../runs" CACHE PATH "Isolated test directories")
set(QCOPILOTS_TOOL_SEMANTIC_MATCHING_MODEL_DIR "" CACHE PATH "Verified text model directory")
set(FETCHCONTENT_BASE_DIR "${QCOPILOTS_TOOL_SEMANTIC_MATCHING_DEPENDENCY_ROOT}" CACHE PATH "External dependency root")
foreach(_root QCOPILOTS_TOOL_SEMANTIC_MATCHING_DEPENDENCY_ROOT QCOPILOTS_TOOL_SEMANTIC_MATCHING_TEST_ROOT FETCHCONTENT_BASE_DIR)
  file(REAL_PATH "${${_root}}" _resolved)
  cmake_path(IS_PREFIX _qctsm_source "${_resolved}" NORMALIZE _inside)
  if(_inside)
    message(FATAL_ERROR "${_root} must stay outside the source tree")
  endif()
endforeach()
file(MAKE_DIRECTORY "${QCOPILOTS_TOOL_SEMANTIC_MATCHING_TEST_ROOT}")
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
if(QCOPILOTS_TOOL_SEMANTIC_MATCHING_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
endif()
include(qctsm-sqlite)
add_subdirectory(src/core)
add_subdirectory(src/provider_http)

if(NOT QCOPILOTS_TOOL_SEMANTIC_MATCHING_CORE_ONLY)
  find_package(Qt6 6.12 REQUIRED COMPONENTS Gui Widgets Concurrent)
  if(NOT Qt6Core_VERSION VERSION_EQUAL "6.12.0")
    message(FATAL_ERROR "Standalone builds use the locked Qt 6.12.0 SDK")
  endif()
  include(qctsm-llama)
  add_subdirectory(src/runtime)
  qctsm_finalize_llama(qctsm-runtime)
  if(NOT TARGET httplib::httplib)
    add_library(qctsm-httplib STATIC "${QCOPILOTS_TOOL_SEMANTIC_MATCHING_LLAMA_SOURCE_DIR}/vendor/cpp-httplib/httplib.cpp")
    set_target_properties(qctsm-httplib PROPERTIES AUTOMOC OFF AUTORCC OFF POSITION_INDEPENDENT_CODE ON)
    target_include_directories(qctsm-httplib SYSTEM PUBLIC "${QCOPILOTS_TOOL_SEMANTIC_MATCHING_LLAMA_SOURCE_DIR}/vendor/cpp-httplib")
    if(WIN32)
      target_link_libraries(qctsm-httplib PUBLIC ws2_32 crypt32)
    endif()
    add_library(httplib::httplib ALIAS qctsm-httplib)
  endif()
  foreach(_module worker service app cli)
    add_subdirectory("src/${_module}")
  endforeach()
endif()

if(QCOPILOTS_TOOL_SEMANTIC_MATCHING_BUILD_TESTS)
  enable_testing()
  file(WRITE "${CMAKE_BINARY_DIR}/CTestCustom.cmake" "set(CTEST_CUSTOM_MAXIMUM_PASSED_TEST_OUTPUT_SIZE 16777216)\nset(CTEST_CUSTOM_MAXIMUM_FAILED_TEST_OUTPUT_SIZE 16777216)\n")
  foreach(_test core runtime service quality)
    if(QCOPILOTS_TOOL_SEMANTIC_MATCHING_CORE_ONLY AND NOT _test STREQUAL "core")
      continue()
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${_test}/CMakeLists.txt")
      add_subdirectory("tests/${_test}")
    endif()
  endforeach()
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME qctsm-tooling COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tools/qctsm-tooling-test.py")
  set_tests_properties(qctsm-tooling PROPERTIES LABELS "unit;tooling")
  add_test(NAME qctsm-quality-contract COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/tests/quality" -p test_quality_contract.py)
  set_tests_properties(qctsm-quality-contract PROPERTIES LABELS "unit;quality" TIMEOUT 60)
endif()

if(NOT QCOPILOTS_TOOL_SEMANTIC_MATCHING_CORE_ONLY)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION "${CMAKE_INSTALL_DATADIR}/qctsm")
  install(FILES config/qctsm-models.lock.json dependencies.lock.json config/build-variants.json DESTINATION "${CMAKE_INSTALL_DATADIR}/qctsm/config")
  include(qctsm-package)
endif()
